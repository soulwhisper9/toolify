<script>
const {PDFDocument, PageSizes} = PDFLib;
let files = [], pdfBytes = null, dragIdx = null;
const dropZone = document.getElementById('dropZone');
const fileInput = document.getElementById('fileInput');
const fmtSize = b => b < 1048576 ? (b/1024).toFixed(1)+' KB' : (b/1048576).toFixed(2)+' MB';

dropZone.addEventListener('dragover', e => {e.preventDefault();dropZone.classList.add('drag-over')});
dropZone.addEventListener('dragleave', () => dropZone.classList.remove('drag-over'));
dropZone.addEventListener('drop', e => {e.preventDefault();dropZone.classList.remove('drag-over');addFiles(e.dataTransfer.files)});
fileInput.addEventListener('change', () => addFiles(fileInput.files));

function addFiles(newFiles) {
  for (const f of newFiles) if (f.type.startsWith('image/')) files.push(f);
  renderList();
}
function renderList() {
  if (!files.length) return;
  document.getElementById('fileList').classList.add('show');
  document.getElementById('optionsRow').style.display = 'flex';
  document.getElementById('actions').style.display = 'flex';
  document.getElementById('totalSize').textContent = `${files.length} images · ${fmtSize(files.reduce((s,f)=>s+f.size,0))}`;
  document.getElementById('fileItems').innerHTML = files.map((f,i) => `
    <div class="file-item" draggable="true" data-i="${i}" ondragstart="dragStart(event,${i})" ondragover="event.preventDefault()" ondrop="dragDrop(event,${i})">
      <span class="drag-handle">⠿</span><span class="order-num">${i+1}</span>
      <span class="file-icon">🖼️</span><span class="file-name">${f.name}</span>
      <span class="file-size">${fmtSize(f.size)}</span>
      <button class="file-remove" onclick="removeFile(${i})">✕</button>
    </div>`).join('');
}
function dragStart(e,i){dragIdx=i;e.currentTarget.classList.add('dragging')}
function dragDrop(e,i){e.preventDefault();if(dragIdx===null||dragIdx===i)return;files.splice(i,0,files.splice(dragIdx,1)[0]);dragIdx=null;renderList()}
function removeFile(i){files.splice(i,1);files.length?renderList():resetAll()}

document.getElementById('convertBtn').addEventListener('click', async () => {
  if(!files.length)return alert('Add at least 1 image.');
  const pw=document.getElementById('progressWrap'), pf=document.getElementById('progressFill');
  pw.classList.add('show');
  document.getElementById('convertBtn').disabled=true;
  try {
    const pageSize = document.getElementById('pageSize').value;
    const orientation = document.getElementById('orientation').value;
    const margin = parseInt(document.getElementById('margin').value);
    const pdf = await PDFDocument.create();
    for(let i=0;i<files.length;i++){
      pf.style.width=Math.round((i/files.length)*90)+'%';
      const bytes = await files[i].arrayBuffer();
      let img;
      if(files[i].type==='image/png') img = await pdf.embedPng(bytes);
      else if(files[i].type==='image/jpeg') img = await pdf.embedJpg(bytes);
      else {
        const dataUrl = await new Promise(r => {
          const im = new Image();
          im.onload = () => {
            const c = document.createElement('canvas');
            c.width = im.width; c.height = im.height;
            c.getContext('2d').drawImage(im, 0, 0);
            r(c.toDataURL('image/png'));
          };
          im.src = URL.createObjectURL(files[i]);
        });
        const b64 = dataUrl.split(',')[1];
        const bin = atob(b64);
        const arr = new Uint8Array(bin.length);
        for(let j=0;j<bin.length;j++) arr[j]=bin.charCodeAt(j);
        img = await pdf.embedPng(arr);
      }
      let pw2, ph;
      if(pageSize==='Fit'){pw2=img.width+margin*2;ph=img.height+margin*2;}
      else {
        const sizes = {A4:PageSizes.A4, Letter:PageSizes.Letter, Legal:PageSizes.Legal};
        [pw2, ph] = sizes[pageSize];
        if(orientation==='landscape' || (orientation==='auto' && img.width>img.height)) [pw2,ph]=[ph,pw2];
      }
      const page = pdf.addPage([pw2, ph]);
      const maxW = pw2 - margin*2, maxH = ph - margin*2;
      const scale = Math.min(maxW/img.width, maxH/img.height, 1);
      const dw = img.width*scale, dh = img.height*scale;
      page.drawImage(img, {x:(pw2-dw)/2, y:(ph-dh)/2, width:dw, height:dh});
    }
    pf.style.width='100%';
    pdfBytes = await pdf.save();
    document.getElementById('resultInfo').textContent = `${files.length} images · ${pdf.getPageCount()} pages · ${fmtSize(pdfBytes.length)}`;
    document.getElementById('resultBox').classList.add('show');
    document.getElementById('shareWa').href = `https://wa.me/?text=Convert JPG to PDF free at Toolify Core: https://toolifycore.com/tools/jpg-to-pdf.html`;
    pw.classList.remove('show');
  } catch(e) {
    console.error(e);
    alert('Error converting images. Please try smaller or fewer files.');
    pw.classList.remove('show');
    document.getElementById('convertBtn').disabled=false;
  }
});
document.getElementById('downloadBtn').addEventListener('click', () => {
  const a=document.createElement('a');
  a.href=URL.createObjectURL(new Blob([pdfBytes],{type:'application/pdf'}));
  a.download='images-toolifycore.pdf';a.click();
});
function resetAll(){files=[];pdfBytes=null;fileInput.value='';
  document.getElementById('fileList').classList.remove('show');
  document.getElementById('optionsRow').style.display='none';
  document.getElementById('actions').style.display='none';
  document.getElementById('resultBox').classList.remove('show');
  document.getElementById('progressFill').style.width='0%';
  document.getElementById('convertBtn').disabled=false;}

function copyEmbed(){navigator.clipboard.writeText(document.querySelector('.embed-code').textContent);alert('Embed code copied!')}
