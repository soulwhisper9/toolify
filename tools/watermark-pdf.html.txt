<script>
const {PDFDocument, rgb, degrees, StandardFonts} = PDFLib;
let pdfFile = null, resultBytes = null;
const dropZone = document.getElementById('dropZone');
const fileInput = document.getElementById('fileInput');
const fmtSize = b => b < 1048576 ? (b/1024).toFixed(1)+' KB' : (b/1048576).toFixed(2)+' MB';

dropZone.addEventListener('dragover', e => {e.preventDefault();dropZone.classList.add('drag-over')});
dropZone.addEventListener('dragleave', () => dropZone.classList.remove('drag-over'));
dropZone.addEventListener('drop', e => {e.preventDefault();dropZone.classList.remove('drag-over');handleFile(e.dataTransfer.files[0])});
fileInput.addEventListener('change', () => handleFile(fileInput.files[0]));

['wmSize','wmOpacity','wmRotation'].forEach(id => {
  document.getElementById(id).addEventListener('input', e => {
    const val = e.target.value;
    if(id==='wmSize') document.getElementById('sizeVal').textContent = val;
    if(id==='wmOpacity') document.getElementById('opacityVal').textContent = val+'%';
    if(id==='wmRotation') document.getElementById('rotVal').textContent = val+'°';
  });
});

function handleFile(f) {
  if(!f || f.type !== 'application/pdf') return alert('Please choose a PDF file.');
  pdfFile = f;
  document.getElementById('optionsGrid').style.display = 'grid';
  document.getElementById('actions').style.display = 'flex';
}
function hexToRgb(hex){const b=parseInt(hex.slice(1),16);return {r:((b>>16)&255)/255, g:((b>>8)&255)/255, b:(b&255)/255}}

document.getElementById('applyBtn').addEventListener('click', async () => {
  if(!pdfFile) return;
  const pw = document.getElementById('progressWrap'), pf = document.getElementById('progressFill');
  pw.classList.add('show');
  document.getElementById('applyBtn').disabled = true;
  try {
    const text = document.getElementById('wmText').value || 'WATERMARK';
    const size = parseInt(document.getElementById('wmSize').value);
    const opacity = parseInt(document.getElementById('wmOpacity').value)/100;
    const rotation = parseInt(document.getElementById('wmRotation').value);
    const color = hexToRgb(document.getElementById('wmColor').value);
    const position = document.getElementById('wmPosition').value;
    const pdf = await PDFDocument.load(await pdfFile.arrayBuffer());
    const font = await pdf.embedFont(StandardFonts.HelveticaBold);
    const pages = pdf.getPages();
    for(let i=0;i<pages.length;i++){
      pf.style.width = Math.round((i/pages.length)*90)+'%';
      const page = pages[i];
      const {width, height} = page.getSize();
      const tw = font.widthOfTextAtSize(text, size);
      const opts = {size, font, color:rgb(color.r,color.g,color.b), opacity, rotate:degrees(rotation)};
      if(position==='center'){
        page.drawText(text, {x:(width-tw)/2, y:height/2, ...opts});
      } else if(position==='top'){
        page.drawText(text, {x:(width-tw)/2, y:height-size-30, ...opts, rotate:degrees(0)});
      } else if(position==='bottom'){
        page.drawText(text, {x:(width-tw)/2, y:30, ...opts, rotate:degrees(0)});
      } else if(position==='diagonal'){
        for(let y=size; y<height+size; y+=size*3){
          for(let x=-size; x<width+size; x+=tw+size*2){
            page.drawText(text, {x, y, ...opts});
          }
        }
      }
    }
    pf.style.width = '100%';
    resultBytes = await pdf.save();
    document.getElementById('resultInfo').textContent = `${pages.length} pages watermarked · ${fmtSize(resultBytes.length)}`;
    document.getElementById('resultBox').classList.add('show');
    document.getElementById('shareWa').href = `https://wa.me/?text=Add watermark to PDF free at Toolify Core: https://toolifycore.com/tools/watermark-pdf.html`;
    pw.classList.remove('show');
  } catch(e) {
    console.error(e);
    alert('Error: Make sure PDF is not password protected.');
    pw.classList.remove('show');
    document.getElementById('applyBtn').disabled = false;
  }
});
document.getElementById('downloadBtn').addEventListener('click', () => {
  const a = document.createElement('a');
  a.href = URL.createObjectURL(new Blob([resultBytes],{type:'application/pdf'}));
  a.download = 'watermarked-toolifycore.pdf'; a.click();
});

  document.getElementById('optionsGrid').style.display='none';
  document.getElementById('actions').style.display='none';
  document.getElementById('resultBox').classList.remove('show');
  document.getElementById('progressFill').style.width='0%';
  document.getElementById('applyBtn').disabled=false;}
